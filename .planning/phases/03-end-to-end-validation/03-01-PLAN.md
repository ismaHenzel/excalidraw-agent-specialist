### Task 2: Author the check script `scripts/e2e_check.sh`

**Deliverable:** `.claude/agents/excalidraw/scripts/e2e_check.sh`, executable (`chmod +x`).

**Structure (template, executor adapts):**
```bash
#!/bin/bash
# End-to-end artefact-side check runner for the Excalidraw closed-loop specialist.
# Validates the loop's outputs after an operator-driven specialist run.
# Modes:
#   success         — assert sibling PNG + verifier-report exist and passed=true; no suffixed siblings.
#   honest-failure  — assert sibling PNG + report exist, passed=false + issues non-empty; no suffixed siblings.
#                      If --final-message <path>, assert it contains LOOP-02 honest-failure tokens
#                      and lacks forbidden success tokens.
# Exit 0 = all asserts hold. Exit 1 = at least one assert failed. Exit 2 = bad invocation.

SCRIPT_DIR="$( cd "$( dirname "${BASH_SOURCE[0]}" )" && pwd )"
AGENT_DIR="$( cd "$SCRIPT_DIR/.." && pwd )"

usage() {
    echo "Usage: bash scripts/e2e_check.sh <path-to-.excalidraw> --mode {success|honest-failure} [--final-message <path>]" >&2
}

# argparse
SOURCE=""
MODE=""
FINAL_MSG=""
while [ "$#" -gt 0 ]; do
    case "$1" in
        --mode)            MODE="$2"; shift 2 ;;
        --final-message)   FINAL_MSG="$2"; shift 2 ;;
        --help|-h)         usage; exit 0 ;;
        *.excalidraw)      SOURCE="$1"; shift ;;
        *)                 usage; exit 2 ;;
    esac
done

[ -z "$SOURCE" ] && { usage; exit 2; }
[ -z "$MODE" ] && { usage; exit 2; }
case "$MODE" in success|honest-failure) ;; *) usage; exit 2 ;; esac
# Validate optional flag had a value
case "$FINAL_MSG" in --*) usage; exit 2 ;; esac

# derive sibling paths
BASE="${SOURCE%.excalidraw}"
PNG="$BASE.png"
REPORT="$BASE.verifier-report.json"
DIR="$(dirname "$SOURCE")"
BASENAME="$(basename "$BASE")"

FAIL=0
pass() { echo "PASS: $1"; }
fail() { echo "FAIL: $1"; FAIL=$((FAIL + 1)); }

# S1/H1 source exists + valid JSON
[ -f "$SOURCE" ] || { fail "source not found: $SOURCE"; echo "Summary: $FAIL fail"; exit 1; }
jq empty "$SOURCE" >/dev/null 2>&1 && pass "source is valid JSON" || fail "source is not valid JSON: $SOURCE"

# S2/H2 PNG exists + non-empty
[ -f "$PNG" ] && [ -s "$PNG" ] && pass "sibling PNG present and non-empty: $PNG" || fail "sibling PNG missing or empty: $PNG"

# S3/H3 report exists + has 5 keys + correct passed value
if [ -f "$REPORT" ] && jq empty "$REPORT" >/dev/null 2>&1; then
    pass "verifier report present and valid JSON"
    if jq -e 'has("passed") and has("checked_at") and has("source") and has("png") and has("issues")' "$REPORT" >/dev/null; then
        pass "verifier report has all 5 top-level keys"
    else
        fail "verifier report missing one of the 5 top-level keys"
    fi
    case "$MODE" in
        success)
            jq -e '.passed == true' "$REPORT" >/dev/null && pass "verifier reports passed=true" || fail "verifier reports passed != true"
            ;;
        honest-failure)
            jq -e '.passed == false' "$REPORT" >/dev/null && pass "verifier reports passed=false" || fail "verifier reports passed != false"
            jq -e '(.issues | length) > 0' "$REPORT" >/dev/null && pass "verifier report has at least one issue" || fail "verifier report has zero issues but mode=honest-failure"
            ;;
    esac

    # S5 soft assert: checked_at not older than source mtime
    CHECKED_AT=$(jq -r '.checked_at // empty' "$REPORT")
    if [ -n "$CHECKED_AT" ]; then
        # parse ISO-8601 to epoch; if `date` can't, warn and skip
        REPORT_EPOCH=$(date -d "$CHECKED_AT" +%s 2>/dev/null || echo "")
        SOURCE_EPOCH=$(stat -c %Y "$SOURCE" 2>/dev/null || echo "")
        if [ -n "$REPORT_EPOCH" ] && [ -n "$SOURCE_EPOCH" ]; then
            if [ "$REPORT_EPOCH" -ge "$SOURCE_EPOCH" ]; then
                pass "report checked_at ($CHECKED_AT) is not older than source mtime"
            else
                fail "STALE: report checked_at ($CHECKED_AT) is older than source mtime — re-run the specialist on this scenario"
            fi
        else
            echo "WARN: could not compare report checked_at to source mtime (parse failure); skipping staleness assert"
        fi
    fi
else
    fail "verifier report missing or not valid JSON: $REPORT"
fi

# S4/H4 artefact discipline — no suffixed siblings
SHOPT_OPTS=$(shopt -p nullglob)  # save state
shopt -s nullglob
BAD=( "$DIR/$BASENAME".v[0-9]*.excalidraw "$DIR/$BASENAME"_iter*.excalidraw "$DIR/$BASENAME"_iter*.png "$DIR/$BASENAME".bak* "$DIR/$BASENAME".report-[0-9]*.json )
eval "$SHOPT_OPTS"  # restore
if [ "${#BAD[@]}" -eq 0 ]; then
    pass "no suffixed siblings (artefact discipline)"
else
    fail "suffixed siblings present (artefact discipline violated): ${BAD[*]}"
fi

# H5 honest-failure final-message asserts
if [ "$MODE" = "honest-failure" ]; then
    if [ -n "$FINAL_MSG" ]; then
        if [ -f "$FINAL_MSG" ] && [ -s "$FINAL_MSG" ]; then
            for TOKEN in "Failed PNG:" "Source:" "Report:" "Outstanding issues:"; do
                if grep -F -q "$TOKEN" "$FINAL_MSG"; then
                    pass "final-message contains honest-failure token: $TOKEN"
                else
                    fail "final-message missing honest-failure token: $TOKEN"
                fi
            done
            # word-bounded, case-insensitive forbidden-token scan
            FORBIDDEN=$(grep -iEn '\b(done|completed|successfully|ready)\b' "$FINAL_MSG" || true)
            if [ -z "$FORBIDDEN" ]; then
                pass "final-message contains no forbidden success tokens"
            else
                fail "final-message contains forbidden success token(s):"; echo "$FORBIDDEN"
            fi
        else
            fail "final-message file missing or empty: $FINAL_MSG"
        fi
    else
        echo "WARN: --final-message not provided; skipping H5 (operator must inspect the message manually)"
    fi
fi

echo "Summary: $FAIL fail"
[ "$FAIL" -eq 0 ] && exit 0 || exit 1
```
